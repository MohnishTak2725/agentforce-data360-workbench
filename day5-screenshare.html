GetAccountMetricsAction.cls 
public without sharing class GetAccountMetricsAction {

    @InvocableMethod(
        label='Get Account Metrics & SLA Tier'
        description='Fetches Closed Won revenue, active ARR (Annual Recurring Revenue), and SLA customer tier for a specified Account. Use this action whenever the user asks for contract value, active ARR, total won revenue, SLA level, or financial tier for a client or company. Do not use for pipeline forecasting, open deals, or support tickets.'
        category='Account Management'
    )
    public static List getMetrics(List requests) {
        List responses = new List();

        for (Request req : requests) {
            Response res = new Response();
            try {
                List accs = new List();

                // Priority 1: Query by Account ID
                if (req.accountId != null) {
                    accs = [
                        SELECT Id, Name, 
                               (SELECT Amount FROM Opportunities WHERE StageName = 'Closed Won') 
                        FROM Account 
                        WHERE Id = :req.accountId 
                        LIMIT 1
                    ];
                } 
                // Priority 2: Query by Account Name
                else if (String.isNotBlank(req.accountName)) {
                    String searchKey = '%' + req.accountName.trim() + '%';
                    accs = [
                        SELECT Id, Name, 
                               (SELECT Amount FROM Opportunities WHERE StageName = 'Closed Won') 
                        FROM Account 
                        WHERE Name LIKE :searchKey 
                        LIMIT 1
                    ];
                }

                if (accs.isEmpty()) {
                    res.isSuccess = false;
                    res.activeARR = 0;
                    res.customerTier = 'Unknown';
                    res.outcomeMessage = 'No matching active Account found in Salesforce for the provided details. Please verify the account name or ID.';
                } else {
                    Account acc = accs[0];
                    Decimal total = 0;
                    for (Opportunity opp : acc.Opportunities) { 
                        total += (opp.Amount != null ? opp.Amount : 0); 
                    }

                    res.isSuccess = true;
                    res.activeARR = total;
                    res.customerTier = (total > 100000) ? 'Platinum' : 'Standard';
                    res.outcomeMessage = 'Account "' + acc.Name + '" has an active ARR of $' + total.format() + 
                                         ' based on Closed Won deals and qualifies for the ' + res.customerTier + ' SLA Tier.';
                }
            } catch (Exception e) {
                res.isSuccess = false;
                res.activeARR = 0;
                res.customerTier = 'Error';
                res.outcomeMessage = 'Unable to compute account metrics due to an internal error: ' + e.getMessage();
            }

            responses.add(res);
        }

        return responses;
    }

    public class Request {
        @InvocableVariable(
            label='Account ID'
            description='The 18-character Salesforce Account record ID (e.g., 001xx000003DGYrAAO). Provide this if running on an Account record page or if the exact ID is present in conversation context.'
            required=false
        )
        public Id accountId;

        @InvocableVariable(
            label='Account Name'
            description='The commercial or company business name of the client (e.g., "Edge Communications"). Use this when the user refers to an account by name in conversational chat.'
            required=false
        )
        public String accountName;
    }

    public class Response { 
        @InvocableVariable(label='Is Success')
        public Boolean isSuccess; 

        @InvocableVariable(label='Outcome Message')
        public String outcomeMessage;

        @InvocableVariable(label='Active ARR')
        public Decimal activeARR;

        @InvocableVariable(label='Customer SLA Tier')
        public String customerTier;
    }
}

⚠️
Crucial Least-Privilege Requirement: "Agentforce Service Agents run under the headless 'Einstein Service Agent User'. It has NO read access to Account/Opportunity by default! Without this step, Atlas fails with 'sObject type Account is not supported'."

  2. Under Object Settings:
    • Select Accounts ➔ Edit ➔ Check Read ➔ Save
    • Select Opportunities ➔ Edit ➔ Check Read ➔ Save
  3. Under Apex Class Access: Move GetAccountMetricsAction to Enabled ➔ Save.
  4. Click Manage Assignments ➔ Add Assignment ➔ Select Einstein Service Agent User ➔ Assign ➔ Done.
